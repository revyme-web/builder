function __applyInstanceSize(variants, w, h) {
  if (w === undefined && h === undefined) return variants;
  const out = {};
  for (const k in variants) {
    out[k] = { ...variants[k], ...(w !== undefined ? { width: w } : {}), ...(h !== undefined ? { height: h } : {}) };
  }
  return out;
}
